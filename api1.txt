<?php
// api.php
include 'db.php';
header('Content-Type: application/json; charset=utf-8');
header('Cache-Control: no-store, no-cache, must-revalidate, max-age=0');
header('Pragma: no-cache');

date_default_timezone_set('Asia/Ho_Chi_Minh');

$action =$_GET['action'] ?? '';

// ==============================================================================
// 1. XỬ LÝ LƯU / XÓA GHI CHÚ NGÀY CHO 2 BÉ (api.php?action=save_daily_note)
// ==============================================================================
if ($action === 'save_daily_note') {$input = json_decode(file_get_contents('php://input'), true);
    $be_id = trim($input['be_id'] ?? '');
    $day_key = trim($input['day_key'] ?? '');
    $content = trim($input['content'] ?? '');

    if (empty($be_id) \vert{}\vert{} empty($day_key)) {
        echo json_encode(['success' => false, 'message' => 'Dữ liệu không hợp lệ']);
        exit;
    }

    try {
        // Tự động khởi tạo bảng daily_notes nếu chưa có
        $conn->exec("CREATE TABLE IF NOT EXISTS `daily_notes` (
            `id` INT AUTO_INCREMENT PRIMARY KEY,
            `be_id` VARCHAR(10) NOT NULL,
            `day_key` VARCHAR(10) NOT NULL,
            `content` TEXT NOT NULL,
            `updated_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
            UNIQUE KEY `uk_be_day` (`be_id`, `day_key`)
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;");

        if (empty($content)) {
            // Chuỗi rỗng -> Xóa ghi chú của ngày đó
            $stmt =$conn->prepare("DELETE FROM `daily_notes` WHERE `be_id` = ? AND `day_key` = ?");
            $stmt->execute([$be_id,$day_key]);
        } else {
            // Thêm mới hoặc cập nhật nếu đã tồn tại
            $stmt =$conn->prepare("INSERT INTO `daily_notes` (`be_id`, `day_key`, `content`) 
                VALUES (?, ?, ?) 
                ON DUPLICATE KEY UPDATE `content` = VALUES(`content`)");
            $stmt->execute([$be_id, $day_key,$content]);
        }

        echo json_encode(['success' => true, 'message' => 'Lưu ghi chú ngày thành công']);
        exit;
    } catch (Exception $e) {
        echo json_encode(['success' => false, 'error' => $e->getMessage()]);
        exit;
    }
}

// ==============================================================================
// 2. XỬ LÝ ĐỒNG BỘ GHI CHÚ / HUY HIỆU TIẾT HỌC LÊN ĐÁM MÂY (api.php?action=sync_todo)
// ==============================================================================
if ($action === 'sync_todo') {$input = json_decode(file_get_contents('php://input'), true);
    $tags =$input['todo_tags'] ?? null;

    if (is_array($tags)) {
        try {
            // Tự động tạo bảng nếu chưa có trong Database
            $conn->exec("CREATE TABLE IF NOT EXISTS `schedule_todos` (
                `slot_key` VARCHAR(100) NOT NULL,
                `type` VARCHAR(20) NOT NULL,
                `content` VARCHAR(255) NOT NULL,
                `expire_at` BIGINT(20) NOT NULL,
                `updated_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
                PRIMARY KEY (`slot_key`)
            ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;");

            // Làm sạch bảng và lưu danh sách mới nhất
            $conn->exec("TRUNCATE TABLE `schedule_todos`");

            if (!empty($tags)) {
                $stmt =$conn->prepare("INSERT INTO `schedule_todos` (`slot_key`, `type`, `content`, `expire_at`) VALUES (?, ?, ?, ?)");
                foreach ($tags as $key =>$val) {
                    if (!empty($val['content'])) {$stmt->execute([
                            (string)$key,
                            (string)($val['type'] ?? 'btvn'),
                            (string)$val['content'],
                            (int)($val['expireAt'] ?? 0)
                        ]);
                    }
                }
            }
            echo json_encode(['success' => true, 'message' => 'Đồng bộ huy hiệu thành công']);
            exit;
        } catch (Exception $e) {
            echo json_encode(['success' => false, 'error' => $e->getMessage()]);
            exit;
        }
    }
    echo json_encode(['success' => false, 'error' => 'Dữ liệu không hợp lệ']);
    exit;
}

// ==============================================================================
// 3. XỬ LÝ ĐỒNG BỘ LỊCH THI TẬP TRUNG LÊN ĐÁM MÂY (api.php?action=sync_exams)
// ==============================================================================
if ($action === 'sync_exams') {$input = json_decode(file_get_contents('php://input'), true);
    $exams =$input['exams'] ?? null;

    if (is_array($exams)) {
        try {
            $conn->exec("CREATE TABLE IF NOT EXISTS `schedule_exams` (
                `id` VARCHAR(50) NOT NULL PRIMARY KEY,
                `be_id` VARCHAR(10) NOT NULL,
                `exam_type` VARCHAR(20) NOT NULL,
                `subject` VARCHAR(100) NOT NULL,
                `exam_date` DATE NOT NULL,
                `start_time` VARCHAR(10) NOT NULL,
                `duration` INT DEFAULT 60,
                `room` VARCHAR(50) DEFAULT '',
                `sbd` VARCHAR(50) DEFAULT '',
                `notes` TEXT DEFAULT NULL,
                `updated_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
            ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;");

            $conn->exec("TRUNCATE TABLE `schedule_exams`");

            if (!empty($exams)) {
                $stmt =$conn->prepare("INSERT INTO `schedule_exams` 
                    (`id`, `be_id`, `exam_type`, `subject`, `exam_date`, `start_time`, `duration`, `room`, `sbd`, `notes`) 
                    VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)");
                foreach ($exams as$ex) {
                    if (!empty($ex['subject']) && !empty($ex['exam_date'])) {$stmt->execute([
                            (string)$ex['id'],
                            (string)($ex['be_id'] ?? '1'),                             (string)($ex['exam_type'] ?? 'gk'),
                            (string)$ex['subject'],
                            (string)$ex['exam_date'],
                            (string)($ex['start_time'] ?? '07:30'),
                            (int)($ex['duration'] ?? 60),                             (string)($ex['room'] ?? ''),
                            (string)($ex['sbd'] ?? ''),
                            (string)($ex['notes'] ?? '')
                        ]);
                    }
                }
            }
            echo json_encode(['success' => true, 'message' => 'Đồng bộ lịch thi thành công']);
            exit;
        } catch (Exception $e) {
            echo json_encode(['success' => false, 'error' => $e->getMessage()]);
            exit;
        }
    }
    echo json_encode(['success' => false, 'error' => 'Dữ liệu không hợp lệ']);
    exit;
}

// ==============================================================================
// 4. TRẢ DỮ LIỆU ĐỒNG BỘ ĐẦY ĐỦ CHO GIAO DIỆN (GET MẶC ĐỊNH)
// ==============================================================================

// 4.1 Cấu hình hệ thống
$settings = [
    'kid1_name'   => 'Trâm Anh',
    'kid1_class'  => '6A3',
    'kid2_name'   => 'Thành Phát',
    'kid2_class'  => '10A4',
    'semester'    => 'HK1',
    'school_year' => '2026-2027'
];
try {
    $st_set =$conn->query("SELECT setting_key, setting_val FROM app_settings");
    if ($st_set) {$db_settings = $st_set->fetchAll(PDO::FETCH_KEY_PAIR);$settings = array_merge($settings,$db_settings);
    }
} catch (Exception $e) {}

// 4.2 Danh sách giáo viên
$teachers_map = [];
try {
    $stmt_tc =$conn->query("SELECT * FROM teachers ORDER BY be_name ASC, subject_name ASC");
    $all_teachers =$stmt_tc->fetchAll(PDO::FETCH_ASSOC);
    foreach ($all_teachers as $tc) {$teachers_map[$tc['id']] =$tc;
    }
} catch (Exception $e) {}

// 4.3 Dữ liệu Thời khóa biểu
$raw_schedules = [];
try {
    $stmt =$conn->query("SELECT * FROM schedule ORDER BY FIELD(buoi, 'SÁNG', 'CHIỀU', 'TỐI'), id ASC");
    if ($stmt) {
        $raw_schedules =$stmt->fetchAll(PDO::FETCH_ASSOC);
    }
} catch (Exception $e) {$raw_schedules = [];
}

$days_keys = ['thu2', 'thu3', 'thu4', 'thu5', 'thu6', 'thu7', 'cn'];$app_data = [];

foreach ($raw_schedules as$row) {
    $be_id = (string)$row['be_name'];
    foreach ($days_keys as $dk) {$raw_content = trim($row[$dk] ?? '');
        if (!empty($raw_content) &&$raw_content !== '-') {
            $lines = array_values(array_filter(explode("\n", str_replace("\r", "", $raw_content))));
            $subject = trim($lines[0] ?? '');
            $note = isset($lines[1]) ? trim(implode(' ', array_slice($lines, 1))) : '';

            $teacher_id = $row['t_' .$dk] ?? null;
            $teacher_info = ($teacher_id && isset($teachers_map[$teacher_id])) ? $teachers_map[$teacher_id] : null;

            $app_data[] = [
                'id'          => $row['id'],
                'be_id'       => $be_id,
                'day'         => $dk,
                'buoi'        => $row['buoi'],
                'tiet'        => $row['tiet'],
                'subject'     => $subject,
                'note'        => $note,
                'teacher'     => $teacher_info
            ];
        }
    }
}

// 4.4 Danh sách huy hiệu bài tập / kiểm tra (schedule_todos)
$cloud_todos = [];
try {
    $now = round(microtime(true) * 1000);
    // Tự động dọn dẹp các huy hiệu đã hết hạn
    $conn->exec("DELETE FROM `schedule_todos` WHERE `expire_at` > 0 AND `expire_at` < $now");

    $st_todo =$conn->query("SELECT * FROM `schedule_todos`");
    if ($st_todo) {
        while ($r =$st_todo->fetch(PDO::FETCH_ASSOC)) {
            $cloud_todos[$r['slot_key']] = [
                'type'     => $r['type'],
                'content'  => $r['content'],
                'expireAt' => (float)$r['expire_at']
            ];
        }
    }
} catch (Exception $e) {$cloud_todos = [];
}

// 4.5 Lấy danh sách Ghi chú ngày (daily_notes)
$daily_notes = [];
try {
    $st_dn =$conn->query("SELECT be_id, day_key, content FROM `daily_notes`");
    if ($st_dn) {
        $daily_notes =$st_dn->fetchAll(PDO::FETCH_ASSOC);
    }
} catch (Exception $e) {$daily_notes = [];
}

// 4.6 Lấy danh sách Lịch thi tập trung (schedule_exams)
$exam_schedules = [];
try {
    $st_ex =$conn->query("SELECT * FROM `schedule_exams` ORDER BY `exam_date` ASC, `start_time` ASC");
    if ($st_ex) {
        $exam_schedules =$st_ex->fetchAll(PDO::FETCH_ASSOC);
    }
} catch (Exception $e) {$exam_schedules = [];
}

// Trả về JSON tổng hợp
echo json_encode([
    'settings'        => $settings,
    'teachers_map'    => $teachers_map,
    'raw_schedules'   => $raw_schedules,
    'schedule_data'   => $app_data,
    'todo_tags'       => $cloud_todos,
    'daily_notes'     => $daily_notes,
    'exam_schedules'  => $exam_schedules,
    'server_time'     => time()
], JSON_UNESCAPED_UNICODE);
exit;
