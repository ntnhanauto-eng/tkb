    $stmt = $conn->prepare("SELECT * FROM schedule WHERE id = ?");
    $stmt->execute([$move_id]);
    $curr = $stmt->fetch(PDO::FETCH_ASSOC);

    if ($curr) {
        $c_buoi = $curr['buoi'];

        $stmt_all = $conn->prepare("SELECT id FROM schedule WHERE be_name = ? AND buoi = ? ORDER BY id ASC");

        $id_list = $stmt_all->fetchAll(PDO::FETCH_COLUMN);

        $pos = array_search($move_id, $id_list);

        if ($pos !== false) {
            $target_pos = ($direction === 'up') ? ($pos - 1) : ($pos + 1);

            if ($target_pos >= 0 && $target_pos < count($id_list)) {
                $target_id = $id_list[$target_pos];

                $stmt_target = $conn->prepare("SELECT * FROM schedule WHERE id = ?");
                $stmt_target->execute([$target_id]);
                $target = $stmt_target->fetch(PDO::FETCH_ASSOC);

                if ($target) {
                    $cols = ['tiet', 'thu2', 'thu3', 'thu4', 'thu5', 'thu6', 'thu7', 'cn', 't_thu2', 't_thu3', 't_thu4', 't_thu5', 't_thu6', 't_thu7', 't_cn'];

                    foreach ($cols as $col) { $params_curr[] = $target[$col] ?? null; }

                    $params_target = [];
                    foreach ($cols as $col) { $params_target[] = $curr[$col] ?? null; }
                    $params_target[] = $target['id'];

// 7. LƯU TOÀN BỘ LỊCH HỌC TRONG TUẦN (TẤT CẢ CÁC THỨ VÀ TIẾT)
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['save_all_schedule'])) {
    $sched_matrix = $_POST['sched'] ?? [];

    foreach ($sched_matrix as $s_id => $days_data) {
        $s_id = intval($s_id);
        
        $sql_parts = [];
        $params = [];

        foreach ($days_map as $d_key => $d_title) {
            $sub = trim($days_data[$d_key]['sub'] ?? '');
            $note = trim($days_data[$d_key]['note'] ?? '');
            $t_id = (!empty($sub) && !empty($days_data[$d_key]['teacher'])) ? intval($days_data[$d_key]['teacher']) : null;

            $cell_text = $sub;
            if (!empty($note)) {
                $cell_text .= "\n" . $note;
            }

            $sql_parts[] = "`{$d_key}` = ?";
            $sql_parts[] = "`t_{$d_key}` = ?";
            $params[] = $cell_text;
            $params[] = $t_id;
        }

        $params[] = $s_id;
        $sql = "UPDATE schedule SET " . implode(', ', $sql_parts) . " WHERE id = ?";

        if (!$is_admin) {
            $sql .= " AND be_name = ?";
            $params[] = $user['be_name'];
        }

        $stmt = $conn->prepare($sql);
        $stmt->execute($params);
    }
    $msg = "Đã lưu toàn bộ lịch học trong tuần thành công!";
}

    // Tự động giữ nguyên vị trí cuộn màn hình khi tải lại trang hoặc submit thao tác
    window.addEventListener('scroll', function() {
        sessionStorage.setItem('admin_scroll_y', window.scrollY);
    });

    window.addEventListener('DOMContentLoaded', function() {
        const savedScroll = sessionStorage.getItem('admin_scroll_y');
        if (savedScroll !== null) {
            window.scrollTo({
                top: parseInt(savedScroll, 10),
                behavior: 'instant'
            });
        }
    });

    window.addEventListener('beforeunload', function() {
        sessionStorage.setItem('admin_scroll_y', window.scrollY);
    });

    // Bật / tắt chế độ Full màn hình (chỉ hoạt động khi quay ngang màn hình)
    function toggleFullscreenMode() {
        const isFull = document.body.classList.toggle('fullscreen-mode');
        const icon = document.getElementById('fullscreenIcon');
        const text = document.getElementById('fullscreenText');

        if (isFull) {
            icon.className = "fa-solid fa-compress";
            text.textContent = "Thoát full";
        } else {
            icon.className = "fa-solid fa-expand";
            text.textContent = "Full màn hình";
        }
    }

    // Bật / tắt hiển thị phần Cấu hình Hệ Thống
    function toggleSettingsSection() {
        const box = document.getElementById('settingsSectionBox');
        const icon = document.getElementById('settingsToggleIcon');
        if (box.style.display === 'none' || box.style.display === '') {
            box.style.display = 'block';
            icon.innerHTML = '<i class="fa-solid fa-chevron-up"></i> Thu gọn';
        } else {
            box.style.display = 'none';
            icon.innerHTML = '<i class="fa-solid fa-chevron-down"></i> Mở ra';
        }
    }

    // Bật / tắt hiển thị phần danh sách giáo viên
    function toggleTeacherSection() {
        const box = document.getElementById('teacherSectionBox');
        const icon = document.getElementById('teacherToggleIcon');
        if (box.style.display === 'none' || box.style.display === '') {
            box.style.display = 'block';
            icon.innerHTML = '<i class="fa-solid fa-chevron-up"></i> Thu gọn';
        } else {
            box.style.display = 'none';
            icon.innerHTML = '<i class="fa-solid fa-chevron-down"></i> Mở ra';
        }
    }

    function handleSubjectInput(inputEl, teacherSelectId) {
        const teacherSelect = document.getElementById(teacherSelectId);
        if (teacherSelect) {
            if (inputEl.value.trim() === '') {
                teacherSelect.value = '';
                teacherSelect.disabled = true;
            } else {
                teacherSelect.disabled = false;
            }
        }
    }

    // Đóng mở menu cây bút
    function toggleDropdown(event, dropdownId) {
        event.stopPropagation();
        const targetMenu = document.getElementById(dropdownId);
        const isOpen = targetMenu.classList.contains('show');

        document.querySelectorAll('.action-dropdown-menu').forEach(menu => {
            menu.classList.remove('show');
        });

        if (!isOpen) {
            targetMenu.classList.add('show');
        }
    }

    // Bấm ra ngoài thì đóng các menu dropdown
    document.addEventListener('click', function () {
        document.querySelectorAll('.action-dropdown-menu').forEach(menu => {
            menu.classList.remove('show');
        });
    });

    // Xử lý modal sửa khung giờ / tiết học
    function openEditSlotModal(slot) {
        document.getElementById('edit_slot_id').value = slot.id;
        document.getElementById('edit_slot_buoi').value = slot.buoi;
        document.getElementById('edit_slot_tiet').value = slot.tiet;

        document.querySelectorAll('.action-dropdown-menu').forEach(menu => menu.classList.remove('show'));
        document.getElementById('editSlotModal').style.display = 'flex';
    }

    function closeEditSlotModal(event) {
        if (!event || event.target.id === 'editSlotModal') {
            document.getElementById('editSlotModal').style.display = 'none';
        }
    }

    // Xử lý modal sửa giáo viên
    function openEditModal(teacher) {
        document.getElementById('edit_teacher_id').value = teacher.id;
        document.getElementById('edit_subject_name').value = teacher.subject_name;
        document.getElementById('edit_teacher_name').value = teacher.teacher_name;
        document.getElementById('edit_phone').value = teacher.phone || '';
        document.getElementById('edit_note').value = teacher.note || '';

        document.querySelectorAll('.action-dropdown-menu').forEach(menu => menu.classList.remove('show'));
        document.getElementById('editTeacherModal').style.display = 'flex';
    }

    function closeEditModal(event) {
        if (!event || event.target.id === 'editTeacherModal') {
            document.getElementById('editTeacherModal').style.display = 'none';
        }
    }
